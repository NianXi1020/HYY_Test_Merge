\section{价差定义与OU建模}
\subsection{价差定义}
若数据中已有 \texttt{ETF\_Futures\_Spread} 列可直接使用；否则定义：
\[
\text{spread}_t = ETF_t - \beta \cdot Futures_t,\quad \beta=1
\]

\subsection{OU过程}
连续形式：
\[
dX_t = \theta(\mu - X_t)dt + \sigma dW_t
\]

离散拟合：
\[
X_{t+1}=\alpha + \beta X_t + \varepsilon_t
\]

参数恢复：
\[
\theta = -\frac{\ln(\beta)}{\Delta t},\quad
\mu = \frac{\alpha}{1-\beta},\quad
\sigma=\text{std}(\varepsilon)
\]

其中 $\Delta t$ 取分钟级（如1）。

\subsection{训练集}
初次估计使用前50\%样本：
\begin{lstlisting}
train_size = int(len(df) * 0.5)
\end{lstlisting}

\section{交易信号规则}
\subsection{z-score}
\[
z_t = \frac{spread_t - \mu_t}{\sigma_t}
\]

\subsection{默认阈值}
\[
entry\_z = 2.0,\quad exit\_z = 0.5
\]

\subsection{开仓平仓逻辑}
\begin{itemize}
  \item 若 $z_t > entry\_z$：做空价差（卖ETF，买期货）
  \item 若 $z_t < -entry\_z$：做多价差（买ETF，卖期货）
  \item 若 $|z_t| < exit\_z$：平仓
\end{itemize}

\section{仓位与风控}
\subsection{上限约束}
\begin{itemize}
  \item 最大资金上限：\texttt{10,000,000}
  \item 最大期货仓位：\texttt{8手}
\end{itemize}

\subsection{动态仓位}
仓位与 $|z_t|$ 成比例：
\[
ratio_t = \min\left(\frac{|z_t|}{4}, 1\right),\quad
contracts_t = \lfloor ratio_t \times 8 \rfloor
\]

并保证不突破资金限制与合约上限。
